2	"""Grid-search сканер стратегий.
3	
4	Перебирает декартово произведение параметров SCAN_GRIDS по каждой стратегии,
5	прогоняет бэктест на train (70%) и out-of-sample test (30%) окнах, считает
6	консервативную оценку combined_sharpe = min(train_sharpe, test_sharpe) и
7	пишет результаты в SQLite. Прогресс уходит клиентам по SSE (scan_progress).
8	"""

24	# Статистика последних прогонов: run_id -> {"total", "kept", "filtered",
25	# "finished"}. Нужна API (/api/scan), чтобы UI знал масштаб скана: сколько
26	# комбинаций сгенерировано и сколько отсеялось по SCAN_MIN_TRADES.
27	RUN_STATS = {}
28	RUN_STATS_MAX = 50  # сколько последних прогонов держим в памяти

31	def get_run_stats(run_id):
32	    """Статистика прогона для API; {} если прогон неизвестен."""
33	    return dict(RUN_STATS.get(run_id) or {})

36	def _prune_run_stats():
37	    """Чистим старые прогоны, чтобы RUN_STATS не рос бесконечно."""
38	    if len(RUN_STATS) <= RUN_STATS_MAX:
39	        return
40	    for run_id in list(RUN_STATS)[:len(RUN_STATS) - RUN_STATS_MAX]:
41	        RUN_STATS.pop(run_id, None)

44	def _generate_combinations(strategy, grid):
45	    """Декартово произведение параметров грида -> list[dict].
46	
47	    Если комбинаций больше SCAN_MAX_COMBINATIONS — случайная подвыборка
48	    с seed=42 (воспроизводимый набор при одинаковом гриде).
49	    """
50	    keys = sorted(grid.keys())  # детерминированный порядок ключей
51	    combos = [
52	        dict(zip(keys, values))
53	        for values in itertools.product(*(grid[k] for k in keys))
54	    ]
55	    if len(combos) > config.SCAN_MAX_COMBINATIONS:
56	        combos = random.Random(42).sample(combos, config.SCAN_MAX_COMBINATIONS)
57	    log.info("scan grid %s: %d combinations", strategy, len(combos))
58	    return combos

61	def _normalize_timeframes(timeframes):
62	    """ТФ скана -> список. str (обратная совместимость) -> [str]; None -> дефолт.
63	
64	    Пустой список остаётся пустым: роут POST /api/scan отдаёт на него 400
65	    («At least one timeframe required»), а run_scan просто ничего не считает.
66	    """
67	    if timeframes is None:
68	        return [config.DEFAULT_TIMEFRAME]
69	    if isinstance(timeframes, str):
70	        return [timeframes]
71	    return [str(tf) for tf in timeframes]

74	def build_plan(strategies, grids=None, symbols=None, timeframes=None):
75	    """План скана.
76	
77	    Без symbols/timeframes — раскладка гридов [(strategy, combos)] (объём
78	    скана, перебор стратегий). С ними — плоский список ЗАДАЧ
79	    [(symbol, tf, strategy, params)] в порядке прогона:
80	    symbol -> tf -> strategy -> params (тот же порядок обходит run_scan).
81	
82	    grids — опциональное переопределение дефолтных гридов (custom_grids
83	    из POST /api/scan): {strategy: {param: [values]}}; для стратегий без
84	    переопределения используется config.SCAN_GRIDS.
85	    """
86	    custom = grids or {}
87	    plan = []
88	    for strategy in strategies:
89	        grid = custom.get(strategy) or config.SCAN_GRIDS.get(strategy)
90	        if not grid:
91	            log.warning("scan: unknown strategy %s — пропуск", strategy)
92	            continue
93	        plan.append((strategy, _generate_combinations(strategy, grid)))
94	    if symbols is None or timeframes is None:
95	        return plan
96	    # params — те же dict-объекты, что и в plan: плоский список задач не
97	    # дублирует комбинации, только ссылки на них.
98	    return [
99	        (symbol, tf, strategy, params)
100	        for symbol in symbols
101	        for tf in timeframes
102	        for strategy, combos in plan
103	        for params in combos
104	    ]

107	def plan_totals(symbols, strategies, grids=None, timeframes=None):
108	    """Сколько всего комбинаций будет прогнано (объём скана до старта).
109	
110	    total = symbols × timeframes × Σ combos_per_strategy. Без timeframes
111	    (старые вызовы) считаем один ТФ.
112	    """
113	    return len(build_plan(strategies, grids, symbols=symbols,
114	                          timeframes=_normalize_timeframes(timeframes)))

117	def _metrics(res):
118	    """Метрики прогона run_backtest -> плоский dict для JSON."""
119	    trades = res.get("trades") or []
120	    gross = sum(t.get("pnl", 0) for t in trades if t.get("pnl", 0) > 0)
121	    loss = abs(sum(t.get("pnl", 0) for t in trades if t.get("pnl", 0) < 0))
122	    if loss > 0:
123	        profit_factor = round(gross / loss, 2)
124	    else:
125	        # run_backtest отдаёт только последние 20 сделок — убытков может
126	        # не быть; бесконечность в JSON/SQLite не кладём, каппим.
127	        profit_factor = 999.0 if gross > 0 else 0.0
128	    return {
129	        "sharpe": res.get("sharpe_ratio", 0),
130	        "winrate": res.get("win_rate", 0),
131	        "max_dd": res.get("max_drawdown", 0),
132	        "total_return": res.get("total_return", 0),
133	        "profit_factor": profit_factor,
134	        "trades": res.get("total_trades", 0),
135	    }

138	def _run_single(symbol, tf, strategy_name, params, df_train, df_test):
139	    """Один прогон комбинации: бэктест на train и на test.
140	
141	    df_train/df_test — готовые окна (уже нарезаны в run_scan): передаются
142	    прямо в run_backtest(df=...), который НЕ ходит за данными. Так 90
143	    комбинаций × 10 символов дают 10 запросов get_replay_df (по одному
144	    на символ), а не 900. combined_sharpe = min(train, test) —
145	    консервативная оценка: стратегия должна быть хороша на обоих окнах.
146	
147	    Отсев: меньше SCAN_MIN_TRADES сделок хотя бы на одном окне.
148	    Возвращает dict или None.
149	    """
150	    if df_train is None or df_test is None \
151	            or len(df_train) < 2 or len(df_test) < 2:
152	        return None
153	
154	    res_train = run_backtest(symbol, tf, None, None,
155	                             strategy_name, params, df=df_train)
156	    res_test = run_backtest(symbol, tf, None, None,
157	                            strategy_name, params, df=df_test)
158	    if "error" in res_train or "error" in res_test:
159	        log.warning("scan %s %s %s: бэктест вернул ошибку",
160	                    symbol, strategy_name, params)
161	        return None
162	
163	    m_train, m_test = _metrics(res_train), _metrics(res_test)
164	    if min(m_train["trades"], m_test["trades"]) < config.SCAN_MIN_TRADES:
165	        return None
166	
167	    return {
168	        "params": params,
169	        "train": m_train,
170	        "test": m_test,
171	        "combined_sharpe": min(m_train["sharpe"], m_test["sharpe"]),
172	        "total_trades": m_train["trades"] + m_test["trades"],
173	    }

176	def _eta_seconds(start, done, total):
177	    """Оценка оставшегося времени (сек) по средней скорости прогона.
178	
179	    rate = done / elapsed (комбинаций в секунду), eta = (total - done) / rate.
180	    None, если ещё ничего не обработано или прогон уже закончился.
181	    """
182	    if not done or done >= total:
183	        return None
184	    elapsed = time.time() - start
185	    if elapsed <= 0:
186	        return None
187	    rate = done / elapsed
188	    if rate <= 0:
189	        return None
190	    return int(round((total - done) / rate))

193	def _task_key(task):
194	    """Ключ группировки задач скана (symbol, tf, strategy) — см. build_plan."""
195	    return task[:3]

198	def _push_progress(run_id, done, total, current, eta_seconds=None,
199	                   tf_index=None, tf_total=None, current_tf=None,
200	                   current_symbol=None):
201	    """SSE-событие прогресса скана (прогресс не должен ломать сам скан).
202	
203	    eta_seconds добавляется не в каждое событие, а раз в
204	    SCAN_ETA_PUSH_EVERY комбинаций (см. run_scan): UI показывает
205	    «Осталось ~M мин».
206	
207	    ТФ-поля (при прогоне по нескольким ТФ): tf_index/tf_total — номер и
208	    всего ТФ, current_tf/current_symbol — где скан сейчас:
209	    {..., "tf_index": 2, "tf_total": 5, "current_tf": "1H",
210	     "current_symbol": "USDCHF", "eta_seconds": 1800}.
211	    """
212	    try:
213	        data = {"run_id": run_id, "done": done, "total": total,
214	                "current": current}
215	        if eta_seconds is not None:
216	            data["eta_seconds"] = eta_seconds
217	        if tf_total is not None:
218	            data.update(tf_index=tf_index, tf_total=tf_total,
219	                        current_tf=current_tf, current_symbol=current_symbol)
220	        _ws_push("scan_progress", data)
221	    except Exception:  # noqa: BLE001
222	        log.exception("scan_progress push failed")

225	def run_scan(symbols, timeframes, strategies, run_id=None, grids=None):
226	    """Запуск скана: символы × ТФ × стратегии, перебор комбинаций.
227	
228	    timeframes — список ТФ (["15m", "1H"]); str принимается для обратной
229	    совместимости (оборачивается в [str]) — см. _normalize_timeframes.
230	
231	    grids — опциональное переопределение дефолтных гридов (custom_grids).
232	
233	    Для каждой пары (symbol, tf): get_replay_df за SCAN_PERIOD_DAYS — ОДИН
234	    запрос на пару (df переиспользуется всеми комбинациями через df=), сплит
235	    70/30 по времени, далее ThreadPoolExecutor(SCAN_WORKERS) -> _run_single
236	    параллельно; прошедшие отсев результаты -> db_save_scan_result (пишется
237	    tf). Прогресс в SSE: {event: "scan_progress", data: {run_id, done, total,
238	    current, tf_index, tf_total, current_tf, current_symbol}}; раз в
239	    SCAN_ETA_PUSH_EVERY комбинаций в событие добавляется eta_seconds
240	    (остаток прогона, сек). Возвращает run_id.
241	    """
242	    run_id = run_id or uuid.uuid4().hex
243	    start = time.time()  # отсчёт для ETA прогресса (см. _eta_seconds)
244	    to_sec = int(time.time())
245	    from_sec = to_sec - config.SCAN_PERIOD_DAYS * 86400
246	
247	    # Объём для прогресса считаем заранее (генерация комбинаций дешёвая).
248	    # Задачи — плоский план: symbol -> tf -> strategy -> params.
249	    tfs = _normalize_timeframes(timeframes)
250	    tasks = build_plan(strategies, grids, symbols=symbols, timeframes=tfs)
251	    total = len(tasks)
252	    tf_index_by_name = {tf: i for i, tf in enumerate(tfs, start=1)}
253	    done = 0
254	    kept = 0  # комбинаций прошло отсев по SCAN_MIN_TRADES
255	    RUN_STATS[run_id] = {"total": total, "kept": 0, "filtered": 0,
256	                         "finished": False, "tf_index": 0,
257	                         "tf_total": len(tfs), "current_tf": None,
258	                         "current_symbol": None}
259	
260	    df = None
261	    df_key = None  # (symbol, tf) текущего df
262	    # Пачки задач по (symbol, tf, strategy): порядок build_plan гарантирует,
263	    # что groupby отдаёт непрерывные группы комбинаций одной стратегии.
264	    for (symbol, tf, strategy), group in itertools.groupby(tasks,
265	                                                           key=_task_key):
266	        params_list = [task[3] for task in group]
267	        tf_index = tf_index_by_name.get(tf, 1)
268	        # Один запрос данных на пару (symbol, tf): готовый df переиспользуется
269	        # во всех комбинациях (см. _run_single -> run_backtest(df=...)).
270	        if df_key != (symbol, tf):
271	            df = get_replay_df(symbol, tf, from_sec, to_sec,
272	                               limit=config.SCAN_REPLAY_LIMIT)
273	            df_key = (symbol, tf)
274	        if df is None or df.empty or len(df) < 2:
275	            log.warning("scan %s %s: нет данных, пропуск %d комбинаций",
276	                        symbol, tf, len(params_list))
277	            done += len(params_list)
278	            _push_progress(run_id, done, total, f"{symbol}:{tf}:no-data",
279	                           tf_index=tf_index, tf_total=len(tfs),
280	                           current_tf=tf, current_symbol=symbol)
281	            continue
282	
283	        # Сплит 70/30 по времени (df отсортирован по timestamp).
284	        split = int(len(df) * config.SCAN_TRAIN_SPLIT)
285	        df_train = df.iloc[:split].reset_index(drop=True)
286	        df_test = df.iloc[split:].reset_index(drop=True)
287	
288	        with ThreadPoolExecutor(
289	                max_workers=config.SCAN_WORKERS) as ex:
290	            futures = {
291	                ex.submit(_run_single, symbol, tf, strategy,
292	                          params, df_train, df_test): params
293	                for params in params_list
294	            }
295	            for fut in as_completed(futures):
296	                params = futures[fut]
297	                done += 1
298	                try:
299	                    res = fut.result()
300	                except Exception:  # noqa: BLE001 — не роняем весь скан
301	                    log.exception("scan: комбинация %s упала", params)
302	                    res = None
303	                if res:
304	                    kept += 1
305	                    db.db_save_scan_result(
306	                        run_id, symbol, tf, strategy,
307	                        res["params"], res["train"], res["test"],
308	                        res["combined_sharpe"], res["total_trades"])
309	                eta = None
310	                if config.SCAN_ETA_PUSH_EVERY \
311	                        and done % config.SCAN_ETA_PUSH_EVERY == 0:
312	                    eta = _eta_seconds(start, done, total)
313	                if eta is not None:
314	                    # Дублируем ETA в RUN_STATS: поллинг-фолбэк UI
315	                    # (GET /api/scan/<run_id>) видит его без SSE.
316	                    RUN_STATS[run_id]["done"] = done
317	                    RUN_STATS[run_id]["eta_seconds"] = eta
318	                # ТФ-поля тоже в RUN_STATS: без SSE поллинг рисует тот же
319	                # «ТФ i/n: tf · symbol».
320	                RUN_STATS[run_id].update(
321	                    tf_index=tf_index, tf_total=len(tfs), current_tf=tf,
322	                    current_symbol=symbol)
323	                _push_progress(run_id, done, total,
324	                               f"{symbol}:{tf}:{strategy}:{params}",
325	                               eta_seconds=eta, tf_index=tf_index,
326	                               tf_total=len(tfs), current_tf=tf,
327	                               current_symbol=symbol)
328	
329	    _push_progress(run_id, total, total, None,  # финальное событие
330	                   tf_index=len(tfs), tf_total=len(tfs),
331	                   current_tf=tfs[-1] if tfs else None, current_symbol=None)
332	    # update, а не новый dict: сохраняем done/eta_seconds, накопленные в
333	    # прогрессе (их читает поллинг-фолбэк UI, GET /api/scan/<run_id> -> stats).
334	    stats = RUN_STATS.get(run_id) or {}
335	    stats.update({"total": total, "kept": kept, "filtered": total - kept,
336	                  "finished": True})
337	    RUN_STATS[run_id] = stats
338	    _prune_run_stats()
339	    return run_id
